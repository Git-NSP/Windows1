import React, {useState} from 'react'

function App() {
  const [count, setCount] = useState(0);
  
  function incrCountFunction(){
    setCount(count + 1);
  }

  return (
    <div>
      <h1>Entered</h1>
      <button onClick={incrCountFunction}>Incr. Count    Count : {count} </button>
      <h1>Done</h1>
    </div>
  )
}

export default App
